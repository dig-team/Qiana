 85    def extendFromTptp(self, tptpFormula: str) -> None:
 86        """Read the body of a TPTP formula and extend the signature.
 87
 88        Extends the signature with the functions and predicates found in the
 89        formula.
 90
 91        Args:
 92            tptpFormula: The body of a TPTP formula, example: "![X1] : p(f(X1),X1)"
 93        """
 94        for symbol, (arity, isFunction) in parseSymbols(tptpFormula).items():
 95            if symbol[0].isupper(): continue # We don't want to add variables
 96            if symbol in get_special_functions_arities() or symbol == getTruthPredicate(): continue
 97            if isQuoted(symbol): continue # We can't know if it's a quoted function or predicate, so we skip the case. 
 98            if isFunction: self.addFunction(symbol, arity)
 99            else: self.addPredicate(symbol, arity)

101    def _parseFormula(formula : str, formulasNext : bool = False) -> List[Tuple[str, int, bool]]:
102        """
103        Take a formula (for example "![X1] p(f(X1))") and returns a list of tuple containing each symbol along with its arity and a isAFunction boolean
104        """
105      
106        binary_connectives = ["&", "=>", "<=>", "|"]
107        parenthesis_level = 0
108        for i, char in enumerate(formula):
109            if char == "(":
110                parenthesis_level += 1
111            elif char == ")":
112                parenthesis_level -= 1
113            elif parenthesis_level == 0 :
114                for connective in binary_connectives:
115                    if formula[i:].startswith(connective):
116                        assert not formulasNext 
117                        return Signature._parseFormula(formula[:i], formulasNext) + Signature._parseFormula(formula[i+len(connective):], formulasNext)
118        if "]" in formula: 
119            splitformula = formula.split("]", 1)[1].strip()
120            assert splitformula.startswith(":"), f"Expected ':' after ']' in {formula}"
121            formula = splitformula[1:].strip()
122        if not formula : return []
123        
124        # Remove outermost parenthesis
125        while formula[0] == "(":
126            parenthesis_level = 0
127            for i, char in enumerate(formula):
128                if char == "(":
129                    parenthesis_level += 1
130                elif char == ")":
131                    parenthesis_level -= 1
132                if parenthesis_level == 0:
133                    break
134            if i == len(formula) - 1:
135                formula = formula[1:-1].strip()
136            else:
137                break
138
139        # Handle the only unary logical connectives 
140        if formula.startswith("~"):
141            return Signature._parseFormula(formula[1:], True)
142
143        binary_connectives = ["&", "=>", "<=>", "|"]
144        parenthesis_level = 0
145        for i, char in enumerate(formula):
146            if char == "(":
147                parenthesis_level += 1
148            elif char == ")":
149                parenthesis_level -= 1
150            elif parenthesis_level == 0 :
151                for connective in binary_connectives:
152                    if formula[i:].startswith(connective):
153                        assert not formulasNext 
154                        return Signature._parseFormula(formula[:i], formulasNext) + Signature._parseFormula(formula[i+len(connective):], formulasNext)
155
156        # Reach this only if there were no toplevel connectives
157        symbol, arguments = Signature._parseTopLevel(formula)
158
159        # Only variables start with an uppercase char
160        if not symbol[0].islower():
161            assert not arguments
162            return []
163        
164        return [(symbol, len(arguments), formulasNext)] + [tuple for arg in arguments for tuple in Signature._parseFormula(arg, True)]

166    def _parseTopLevel(formula : str) -> Tuple[str, List[str]]:
167            """
168            Read a formula of the form $f(a1, a2, ..., an)$ and return the function name and the arguments
169            """
170            formula = formula.strip()
171            name_end = formula.find('(')
172            if name_end == -1:
173                return formula.strip(), []
174            
175            name = formula[:name_end].strip()
176            
177            if not formula.endswith(')'):
178                raise ValueError(f"Formula missing closing parenthesis: {formula}")
179            
180            arg_string = formula[name_end + 1:-1]
181            
182            args = []
183            if arg_string.strip():
184                current_arg = ""
185                paren_level = 0
186                for char in arg_string:
187                    if char == '(' and paren_level == 0:
188                        current_arg += char
189                        paren_level += 1
190                    elif char == '(' and paren_level > 0:
191                        current_arg += char
192                        paren_level += 1
193                    elif char == ')' and paren_level > 1:
194                        current_arg += char
195                        paren_level -= 1
196                    elif char == ')' and paren_level == 1:
197                        current_arg += char
198                        paren_level -= 1
199                    elif char == ',' and paren_level == 0:
200                        args.append(current_arg.strip())
201                        current_arg = ""
202                    else:
203                        current_arg += char
204                
205                if current_arg:
206                    args.append(current_arg.strip())
207            
208            return name, args
