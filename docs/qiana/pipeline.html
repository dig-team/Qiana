 10def run_qiana(input : str, quotedVariableNumber: int = 5, simplified_input: bool = False, expand_macros: bool = False, timeout: int = 5) -> str:
 11    """Compute the qiana closure of the input and run the solver on it.
 12    This detects contradictions in the qiana closure of the input (which are indicated by the result "unsat").
 13    If a conjecture and axioms are used in the input, the fact that the conjecture follows from the axioms is indicated by such a contradiction.
 14
 15    This function only returns the simple result obtained by running the solver on the qiana closure. If you need more granular control or detailed results, use the QianaPipeline class instead.
 16    This function is a convenience function entirely reliant on the QianaPipeline class.
 17
 18    Args:
 19        input: The tptp representation of a set of formulas, either full
 20            first-order TPTP formulas or simplified syntax.
 21        quotedVariableNumber: Number of quoted variables to use in the qiana
 22            closure. Defaults to 5.
 23        simplified_input: If True, the headers on formulas will be
 24            automatically added. Defaults to False.
 25        expand_macros: If True, the qiana specific macros will be expanded
 26            before computing the qiana closure. Defaults to False.
 27        timeout: The timeout value for the solver in seconds. Defaults to 5.
 28
 29    Returns:
 30        Either "sat", "unsat", "unknown", or "timeout" depending on the result
 31        of the solver. Note that "timeout" is estimated from the time it took
 32        the solver to answer "unknown" and therefore may in rare cases be
 33        output where "unknown" would be more appropriate.
 34
 35    Examples:
 36
 37        >>>  input = "fof(axiom1, axiom, p(a)). \\n fof(conjecture1, conjecture, p(a))."
 38        >>> run_qiana(input, False, False, 5)
 39        'unsat'
 40
 41    Raises:
 42        ValueError: If the solver returns an error or if the input is not a
 43            valid tptp representation of a set of formulas.
 44    """
 45    pipeline = QianaPipeline()
 46    pipeline.compute_qiana_closure(input, quotedVariableNumber, simplified_input, expand_macros)
 47    pipeline.run_compute(timeout, compute_steps=False)
 48    if pipeline.get_solver_result().simpleResult == "error":
 49        raise ValueError(f"Vampire returned an error: {pipeline.get_solver_result().error} \n \n The standard output of the solver was: {pipeline.get_solver_result().vampireOutput}")
 50    return pipeline.get_solver_result().simpleResult

 53    """
 54    Class to handle the entire Qiana pipeline: compute the qiana closure, call the solver, and return the result.
 55    To use this class in order to run computations on a set of formulas, one must instantiate it (calling its constructor) and then call its methods in order.
 56    The typical order of method calls is:
 57
 58    >>> pipeline = QianaPipeline()
 59    >>> pipeline.compute_qiana_closure(input, quotedVariableNumber, simplified_input, expand_macros)
 60    >>> pipeline.run_compute(timeout, compute_steps)
 61    >>> result = pipeline.get_solver_result()
 62    >>> solver_output = pipeline.get_vampire_output()
 63    >>> simple_result = result.simpleResult
 64    """

 67    htmlTree : str | None

 78    def compute_qiana_closure(self, input: str, quotedVariableNumber: int | None = 5, simplified_input : bool = False, expand_macros : bool = False) -> None:
 79        """Compute the qiana closure of the input and store it in self.qianaClosure.
 80
 81        Args:
 82            input: The tptp representation of a set of formulas.
 83            quotedVariableNumber: The number of quoted variables to use in the
 84                qiana closure. Defaults to 5. This argument should ideally
 85                always be specified when calling from the CLI as otherwise the
 86                value stored in Settings will be used.
 87            simplified_input: If True, the headers on formulas will be
 88                automatically added. Defaults to False.
 89            expand_macros: If True, the qiana specific macros will be expanded
 90                before computing the qiana closure. Defaults to False.
 91        """
 92        if simplified_input: input = extend_simplified_syntax(input)
 93        if expand_macros: input = applyMacros(input)
 94
 95        path_to_schemes = join(dirname(__file__), "qianaExtension/qianaAxio.schemes")
 96        with open(path_to_schemes, "r") as f: schemeLines = f.readlines()
 97        if not quotedVariableNumber:
 98            from gui import Settings
 99            quotedVariableNumber = Settings.getQuotedVarsNumber()
100        signature = Signature(nbrQuotedVars=quotedVariableNumber)
101        signature.extendFromTptps(input)
102        # self.qianaClosure : str = os.linesep.join(qianaClosure(input, variableNumber))
103        self.qianaClosure = input + os.linesep + os.linesep.join(getAllSchemesInstances(schemeLines, signature))

112            compute_steps: Whether to compute the reasoning steps or not when
113                a contradiction is found. Defaults to True.
114        """
115        self._callSolver(timeout, compute_steps)

196        """Compute and run the solver for GUI usage.

212    def _callSolver(self, timeout: int, get_reasonin_steps : bool) -> None:
213        """Call the solver and store the result in self.reasoningSteps.
214
215        Assumes the qiana closure has already been computed and stored in
216        self.qianaClosure.
217
218        Args:
219            timeout: The timeout value for the solver.
220            get_reasonin_steps: Whether to compute the reasoning steps or not
221                when a contradiction is found.
222
223        Raises:
224            ValueError: If the qiana closure has not been computed yet.
225        """
226        if not self.qianaClosure: raise ValueError("Qiana closure has not been computed yet. Please call computeQianaClosure() before running the solver.")
227        self.solver_call : SolverCall = SolverCall.call_solver(self.qianaClosure, timeout, get_reasonin_steps)
228        self.foundContradiction = self.solver_call.simpleResult == "unsat"
229        self.simpleResult = self.solver_call.simpleResult
230        self.reasoningSteps = self.solver_call.reasoningSteps
231        self.vampireOutput = self.solver_call.vampireOutput
232        self.htmlTree = getHtmlFromSteps(self.reasoningSteps) if self.foundContradiction else getHtmlNoContradiction(self.vampireOutput)
